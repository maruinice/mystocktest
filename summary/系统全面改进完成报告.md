### 1. 实时数据获取
```typescript
// 批量获取实时行情
const response = await axios.post('/api/quote/realtime/batch', {
  codes: ['000001', '600519', '600036', '000858']
})
marketHotStocks.value = Object.values(response.data.data)
```

### 2. 自动刷新
```typescript
// 每30秒刷新
refreshTimer = window.setInterval(() => {
  fetchMarketHotStocks()
}, 30000)
```

### 3. 数据容错
```typescript
// 确保数据存在
const totalAssets = computed(() => tradingStore.totalAssets || 100000)
const positions = computed(() => 
  Array.isArray(tradingStore.positions) ? tradingStore.positions : []
)
```

### 4. 价格显示优先级
```typescript
// 优先使用实时价格

// 优先使用成交价
row.avg_price || row.price || 0
```
